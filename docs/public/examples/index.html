 8    async sendRequest(method, params = {}) {
 9        return new Promise((resolve, reject) => {
10            const data = JSON.stringify({
11                jsonrpc: '2.0',
12                id: Date.now(),
13                method,
14                params
15            });
16            
17            const options = {

27            
28            const req = http.request(options, (res) => {
29                let body = '';
30                res.on('data', chunk => body += chunk);
31                res.on('end', () => {
32                    try {
33                        resolve(JSON.parse(body));
34                    } catch (e) {
35                        reject(e);
36                    }
37                });
38            });
39            
40            req.on('error', reject);
41            req.write(data);
42            req.end();

44    }

Automation Scripts

Break Reminder Script

 1#!/bin/bash
 2# Auto-trigger break reminder after 1 hour
 3
 4while true; do
 5    DURATION=$(curl -s -X POST http://localhost:8089/mcp \
 6        -H "Content-Type: application/json" \
 7        -d '{"jsonrpc":"2.0","id":1,"method":"get_session_info","params":{}}' \
 8        | jq -r '.result.duration' | sed 's/m.*//')
 9    
10    if [ "$DURATION" -gt 60 ]; then
11        echo "Session too long, triggering break reminder..."
12        curl -s -X POST http://localhost:8089/mcp \
13            -H "Content-Type: application/json" \
14            -d '{"jsonrpc":"2.0","id":2,"method":"trigger_alert","params":{"alert_index":0}}'
15        break
16    fi
17    
18    sleep 300  # Check every 5 minutes
19done

  4class HyperfocusMonitor {
  5    constructor() {
  6        this.alertThresholds = {
  7            'low': 30,      // 30 minutes
  8            'medium': 60,   // 1 hour
  9            'high': 120,    // 2 hours
 10            'critical': 240 // 4 hours
 11        };
 12    }
 13    
 14    async checkHyperfocus() {
 15        try {
 16            const session = await this.getSessionInfo();
 17            const hyperfocus = await this.getHyperfocusStatus();
 18            
 19            if (hyperfocus.result.is_hyperfocus) {
 20                const duration = this.parseDuration(hyperfocus.result.duration);
 21                const level = hyperfocus.result.level;
 22                
 23                console.log(`Hyperfocus detected: ${level} level, ${duration} minutes`);
 24                
 25                // Check if threshold exceeded
 26                if (duration > this.alertThresholds[level]) {
 27                    console.log(`Threshold exceeded for ${level} level!`);
 28                    await this.triggerAlert(this.getAlertIndex(level));
 29                }
 30            }
 31        } catch (error) {
 32            console.error('Error checking hyperfocus:', error);
 33        }
 34    }
 35    
 36    async getSessionInfo() {

 42    }
 43    
 44    async triggerAlert(alertIndex) {
 45        return this.sendRequest('trigger_alert', { alert_index: alertIndex });
 46    }
 47    
 48    sendRequest(method, params = {}) {
 49        return new Promise((resolve, reject) => {
 50            const data = JSON.stringify({
 51                jsonrpc: '2.0',
 52                id: Date.now(),
 53                method,
 54                params
 55            });
 56            
 57            const options = {
 58                hostname: 'localhost',
 59                port: 8089,
 60                path: '/mcp',
 61                method: 'POST',
 62                headers: {
 63                    'Content-Type': 'application/json',
 64                    'Content-Length': Buffer.byteLength(data)
 65                }
 66            };
 67            
 68            const req = http.request(options, (res) => {
 69                let body = '';
 70                res.on('data', chunk => body += chunk);
 71                res.on('end', () => {
 72                    try {
 73                        resolve(JSON.parse(body));
 74                    } catch (e) {
 75                        reject(e);
 76                    }
 77                });
 78            });
 79            
 80            req.on('error', reject);
 81            req.write(data);

 86    parseDuration(duration) {
 87        // Parse duration string like "1h30m" to minutes
 88        const hours = (duration.match(/(\d+)h/) || [0, 0])[1];

Session Data Export

 1import requests
 2import json
 3import csv
 4from datetime import datetime, timedelta
 5
 6class FocusHelperAnalytics:
 7    def __init__(self):
 8        self.base_url = "http://localhost:8089"
 9        self.session = requests.Session()
10        self.session.headers.update({"Content-Type": "application/json"})
11    
12    def export_session_data(self, days=7):
13        """Export session data for the last N days"""
14        data = []
15        
16        for i in range(days):
17            date = datetime.now() - timedelta(days=i)
18            # In a real implementation, you'd query historical data
19            # For now, we'll just get current session info
20            session_info = self.get_session_info()
21            if session_info.get('result'):
22                data.append({
23                    'date': date.strftime('%Y-%m-%d'),
24                    'duration': session_info['result']['duration'],
25                    'is_active': session_info['result']['is_active'],
26                    'hyperfocus_level': session_info['result'].get('hyperfocus_level', ''),
27                    'subject': session_info['result']['subject']
28                })
29        
30        # Export to CSV
31        with open('focus_sessions.csv', 'w', newline='') as csvfile:
32            fieldnames = ['date', 'duration', 'is_active', 'hyperfocus_level', 'subject']
33            writer = csv.DictWriter(csvfile, fieldnames=fieldnames)
34            writer.writeheader()
35            writer.writerows(data)
36        
37        print(f"Exported {len(data)} session records to focus_sessions.csv")
38    
39    def get_session_info(self):
40        response = self.session.post(f"{self.base_url}/mcp", json={
41            "jsonrpc": "2.0",
42            "id": 1,
43            "method": "get_session_info",
44            "params": {}
45        })
46        return response.json()
